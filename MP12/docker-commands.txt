# Build image
docker build -t mp12-image .

# Run image
docker run -it mp12-image

# Login to dockerhub from cli
docker login

# List images
docker image ls

# Tag an image (i.e., tag a locally created image with the same name as a repo established in dockerhub, enabling push [see next command])
docker tag mp12-image bfridthekid/cs498-mp12

# Push an image to dockerhub
docker push bfridthekid/cs498-mp12
